+    // The unguarded update, kept as an oracle for the zero-impulse fast path.
+    fn apply_reference(va: &mut Vw, vb: &mut Vw, d: FPVec3, ima: FP, imb: FP, row: &Row, lam: FP) {
+        if ima != FP::ZERO {
+            let sa = nmul(ima, lam);
+            va.v.x -= nmul(d.x, sa);
+            va.v.y -= nmul(d.y, sa);
+            va.v.z -= nmul(d.z, sa);
+            va.w -= scale(row.wad, lam);
+        }
+        if imb != FP::ZERO {
+            let sb = nmul(imb, lam);
+            vb.v.x += nmul(d.x, sb);
+            vb.v.y += nmul(d.y, sb);
+            vb.v.z += nmul(d.z, sb);
+            vb.w += scale(row.wbd, lam);
+        }
+    }

+    #[test]
+    fn zero_impulse_preserves_all_velocity_bits_at_raw_limits() {
+        let values = [i64::MIN, i64::MIN + 1, -65536, -1, 0, 1, 65536, i64::MAX - 1, i64::MAX];
+        for &raw in &values {
+            assert_eq!(nmul(FP::from_raw(raw), FP::ZERO).raw(), 0);
+        }
+        for index in 0..values.len() {
+            let vector = |offset: usize| FPVec3::new(
+                FP::from_raw(values[(index + offset) % values.len()]),
+                FP::from_raw(values[(index + offset + 1) % values.len()]),
+                FP::from_raw(values[(index + offset + 2) % values.len()]),
+            );
+            let before = [Vw { v: vector(0), w: vector(3) }, Vw { v: vector(6), w: vector(1) }];
+            let row = Row { wad: vector(2), wbd: vector(5), ..Row::default() };
+            for &ima in &values {
+                for &imb in &values {
+                    let [mut va, mut vb] = before;
+                    let [mut reference_a, mut reference_b] = before;
+                    apply(&mut va, &mut vb, vector(4), FP::from_raw(ima), FP::from_raw(imb), &row, FP::ZERO);
+                    apply_reference(&mut reference_a, &mut reference_b, vector(4), FP::from_raw(ima), FP::from_raw(imb), &row, FP::ZERO);
+                    assert_same_velocities(&[va, vb], &before);
+                    assert_same_velocities(&[va, vb], &[reference_a, reference_b]);
+                }
+            }
+        }
+    }
